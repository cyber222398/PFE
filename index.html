Servo servo;

#define penup 115
#define pendown 160
#define Spin 9

//Structures, global variables
struct point {
  float x;
  float y;
  float z;
};

//Current position of plothead
struct point actuatorPos;

//Drawing settings, should be OK
float StepInc = 1;
int StepDelay = 1;
int LineDelay = 0;

//Steps per width and height of the drawing area
float StepsPerMillimeterX = 164.0;
float StepsPerMillimeterY = 250.0;

//Max and min values of the drawing area(mm)
float Xmin = 0;
float Xmax = 38;
float Ymin = 0;
float Ymax = 38;
float Zmin = 0;
float Zmax = 1;

float Xpos = Xmin;
float Ypos = Ymin;
float Zpos = Zmax;

// Set to true to get debug output.
boolean verbose = false;


void setup() {
  Serial.begin( 9600 );
  servo.attach(Spin);
  servo.write(penup);
  delay(100);
  motorx.setSpeed(600);
  motory.setSpeed(600);
  Serial.println("Your CNC machine is ready...");
} 

void loop() {
  delay(100);
  char line[512];
  char c;
  int lineIndex;
  bool lineIsComment, lineSemiColon;

  lineIndex = 0;
  lineSemiColon = false;
  lineIsComment = false;

  while (1) {

    // Serial reception - Mostly from Grbl, added semicolon support
    while ( Serial.available() > 0 ) {
      c = Serial.read();
      if (( c == '\n') || (c == '\r') ) {             
        if ( lineIndex > 0 ) {                        
          line[ lineIndex ] = '\0';                   
          if (verbose) {
            Serial.print( "Received : ");
            Serial.println( line );
          }
          processIncomingLine( line, lineIndex );
          lineIndex = 0;
        }
        else {
          // Empty or comment line. Skip block.
        }
        lineIsComment = false;
        lineSemiColon = false;
        Serial.println("ok");
      }
      else {
        if ( (lineIsComment) || (lineSemiColon) ) {   
          if ( c == ')' )  lineIsComment = false;     
        }
        else {
          if ( c <= ' ' ) {                           
          }
          else if ( c == '/' ) {                    
          }
          else if ( c == '(' ) {                    
            lineIsComment = true;
          }
          else if ( c == ';' ) {
            lineSemiColon = true;
          }
          else if ( lineIndex >= 512 - 1 ) {
            Serial.println( "ERROR - lineBuffer overflow" );
            lineIsComment = false;
            lineSemiColon = false;
          }
          else if ( c >= 'a' && c <= 'z' ) {        
            line[ lineIndex++ ] = c - 'a' + 'A';
          }
          else {
            line[ lineIndex++ ] = c;
          }
        }
      }
    }
  }
}

void processIncomingLine( char* line, int charNB ) {
  int currentIndex = 0;
  char buffer[ 64 ];                                 
  struct point newPos;

  newPos.x = 0.0;
  newPos.y = 0.0;

  while ( currentIndex < charNB ) {
    switch ( line[ currentIndex++ ] ) {              
      case 'U':
        penUp();
        break;
      case 'D':
        penDown();
        break;
      case 'G':
        buffer[0] = line[ currentIndex++ ];          
        buffer[1] = '\0';

        switch ( atoi( buffer ) ) {                  
          case 0:                                   
          case 1:
            char* indexX = strchr( line + currentIndex, 'X' ); 
            char* indexY = strchr( line + currentIndex, 'Y' );
            if ( indexY <= 0 ) {
              newPos.x = atof( indexX + 1);
              newPos.y = actuatorPos.y;
            }
            else if ( indexX <= 0 ) {
              newPos.y = atof( indexY + 1);
              newPos.x = actuatorPos.x;
            }
            else {
              newPos.y = atof( indexY + 1);
              indexY = '\0';
              newPos.x = atof( indexX + 1);
            }
            drawLine(newPos.x, newPos.y );
            actuatorPos.x = newPos.x;
            actuatorPos.y = newPos.y;
            break;
        }
        break;
      case 'M':
        buffer[0] = line[ currentIndex++ ];        
        buffer[1] = line[ currentIndex++ ];
        buffer[2] = line[ currentIndex++ ];
        buffer[3] = '\0';
        switch ( atoi( buffer ) ) {
          case 300:
            {
              char* indexS = strchr( line + currentIndex, 'S' );
              float Spos = atof( indexS + 1);
              if (Spos == 30) {
                penDown();
              }
              if (Spos == 50) {
                penUp();
              }
              break;
            }
          case 114:                                
            Serial.print( "Absolute position : X = " );
            Serial.print( actuatorPos.x );
            Serial.print( "  -  Y = " );
            Serial.println( actuatorPos.y );
            break;
          default:
            Serial.print( "Command not recognized : M");
            Serial.println( buffer );
        }
    }
  }
}

//lines drawing function
void drawLine(float x1, float y1) {

  if (verbose) {
    Serial.print("fx1, fy1: ");
    Serial.print(x1);
    Serial.print(",");
    Serial.print(y1);
    Serial.println("");
  }

  y1 = y1 * -1;

  //  Bring instructions within limits
  if (x1 >= Xmax) {
    x1 = Xmax;
  }
  if (x1 <= Xmin) {
    x1 = Xmin;
  }
  if (y1 >= Ymax) {
    y1 = Ymax;
  }
  if (y1 <= Ymin) {
    y1 = Ymin;
  }

  //  Convert coordinates to steps
  x1 = (int)(x1 * StepsPerMillimeterX / 2);
  y1 = (int)(y1 * StepsPerMillimeterY / 2);
  float x0 = Xpos;
  float y0 = Ypos;

  //  Let's find out the change for the coordinates
  long dx = abs(x1 - x0);
  long dy = abs(y1 - y0);
  int sx = x0 < x1 ? StepInc : -StepInc;
  int sy = y0 < y1 ? StepInc : -StepInc;

  long i;
  long over = 0;

  if (dx > dy) {
    for (i = 0; i < dx; ++i) {
      motorx.onestep(sx, MICROSTEP);
      over += dy;
      if (over >= dx) {
        over -= dx;
        motory.onestep(sy, MICROSTEP);
      }
      delay(StepDelay);
    }
  } else {
    for (i = 0; i < dy; ++i) {
      motory.onestep(sy, MICROSTEP);
      over += dx;
      if (over >= dy) {
        over -= dy;
        motorx.onestep(sx, MICROSTEP);
      }
      delay(StepDelay);
    }
  }

  Xpos = x1;
  Ypos = y1;

}

void penUp() {
  servo.write(penup);
  delay(500);
}

void penDown() {
  servo.write(pendown);
  delay(500);
}

import java.awt.event.KeyEvent;
import javax.swing.JOptionPane;
import processing.serial.*;

Serial port = null;

String portname = null;
boolean streaming = false;
float speed = 0.001;
String[] gcode;
int i = 0;

// Status tracking
String lastResponse = "";
String errorMessage = "";
int okCount = 0;
int errorCount = 0;
boolean waitingResponse = false;
int lastResponseTime = 0;

// UI Colors
color bgColor = #1a1a2e;
color accentColor = #e94560;
color successColor = #4ecca3;
color warningColor = #f39c12;
color textColor = #eaeaea;
color buttonColor = #16213e;
color buttonHover = #0f3460;

boolean portBtnHover = false;
boolean fileBtnHover = false;

void openSerialPort(){
  if (portname == null) return;
  if (port != null) port.stop();
  
  port = new Serial(this, portname, 9600);
  port.bufferUntil('\n');
  okCount = 0;
  errorCount = 0;
  errorMessage = "";
}

void selectSerialPort(){
  String result = (String) JOptionPane.showInputDialog(frame,
    "Select the COM port on the Arduino board",
    "Select serial port",
    JOptionPane.QUESTION_MESSAGE,
    null,
    Serial.list(),
    0);
    
  if (result != null) {
    portname = result;
    openSerialPort();
  }
}

void setup(){
  size(650, 480);
  textFont(createFont("Arial", 12));
  openSerialPort();
}

void draw(){
  background(bgColor);
  
  drawHeader();
  drawStatusPanel();
  drawMotorStatus();
  drawCommunicationLog();
  drawButtons();
  drawProgress();
  drawFooter();
  
  checkHover();
  
  // Check for timeout (no response from Arduino)
  if (waitingResponse && millis() - lastResponseTime > 5000) {
    errorMessage = "Timeout: No response from Arduino";
    waitingResponse = false;
  }
}

  noStroke();
  rect(0, 0, width, 50);

void drawStatusPanel() {
  int y = 65;
  
  fill(textColor);
  textSize(14);
  textAlign(LEFT, TOP);
  
  // Connection status
  fill(150);
  text("CONNECTION", 20, y);
  y += 20;
  
  fill(textColor);
  String statusText = (port != null && portname != null) ? portname : "Not connected";
  text("Port: " + statusText, 20, y);
  y += 18;
  
  // Streaming status
  fill(150);
  text("STATUS", 20, y);
  y += 20;
  
  if (streaming) {
    fill(successColor);
    text(" STREAMING", 20, y);
    fill(150);
    text("  Waiting for: " + gcode[i], 20, y + 18);
  } else if (gcode != null && i >= gcode.length) {
    fill(successColor);
    text(" COMPLETED", 20, y);
  } else {
    fill(150);
    text(" Idle", 20, y);
  }
  
  y += 40;
  
  // Error message
  if (errorMessage.length() > 0) {
    fill(accentColor);
    text("ERROR: " + errorMessage, 20, y);
  }
}

void drawMotorStatus() {
  int x = 300;
  int y = 65;
  
  fill(150);
  textSize(14);
  textAlign(LEFT, TOP);
  text("MOTOR STATUS", x, y);
  y += 25;
  
  // Based on last response from Arduino
  fill(textColor);
  if (lastResponse.length() > 0) {
    if (lastResponse.contains("ok")) {
      fill(successColor);
      text(" Motors responding", x, y);
    } else if (lastResponse.contains("error")) {
      fill(accentColor);
      text(" Motor error detected", x, y);
    } else {
      fill(warningColor);
      text(" Response: " + lastResponse, x, y);
    }
  } else {
    fill(100);
    text(" No data", x, y);
  }
  
  y += 20;
  
  // Communication stats
  fill(150);
  text("COMMUNICATION", x, y);
  y += 20;
  
  fill(successColor);
  text("OK responses: " + okCount, x, y);
  y += 18;
  
  fill(accentColor);
  text("Errors: " + errorCount, x, y);
  y += 18;
  
  if (waitingResponse) {
    fill(warningColor);
    text(" Awaiting response...", x, y);
  }
}

void drawCommunicationLog() {
  int logY = 200;
  
  fill(150);
  textSize(14);
  textAlign(LEFT, TOP);
  text("COMMUNICATION LOG", 20, logY);
  
  // Log background
  fill(20);
  noStroke();
  rect(20, logY + 20, width - 40, 80, 5);
  
  // Log content
  fill(textColor);
  textSize(11);
  textAlign(LEFT, TOP);
  
  // Show last few serial messages
  String logText = lastResponse.length() > 0 ? lastResponse : "Waiting for serial data...";
  
  // Truncate if too long
  if (logText.length() > 60) {
    logText = logText.substring(0, 57) + "...";
  }
  
  text("Last: " + logText, 30, logY + 25);
  
  if (streaming && gcode != null) {
    text("Sent: " + gcode[i], 30, logY + 45);
  }
  
  text("Total lines: " + (gcode != null ? gcode.length : 0), 30, logY + 65);
  text("Processed: " + i, 30, logY + 85);
}

void checkHover() {
  portBtnHover = mouseX > 20 && mouseX < 160 && mouseY > 310 && mouseY < 345;
  fileBtnHover = mouseX > 175 && mouseX < 315 && mouseY > 310 && mouseY < 345;
  stopBtnHover = mouseX > 330 && mouseX < 470 && mouseY > 310 && mouseY < 345;
}

void mousePressed() {
  if (portBtnHover) {
    selectSerialPort();
  }
  
  if (fileBtnHover && !streaming) {
    gcode = null; i = 0;
    okCount = 0;
    errorCount = 0;
    errorMessage = "";
    selectInput("Select a G-code file to process:", "fileSelected", null);
  }
  
  if (stopBtnHover) {
    streaming = false;
    waitingResponse = false;
  }
  
  // Clear errors button
  if (mouseX > width - 160 && mouseX < width - 20 && mouseY > 310 && mouseY < 345) {
    errorMessage = "";
    errorCount = 0;
  }
}

void keyPressed(){
  if (!streaming) {
    if (key == 'p' || key == 'P') selectSerialPort();
  }
  
  if (!streaming && (key == 'g' || key == 'G')) {
    gcode = null; i = 0;
    okCount = 0;
    errorCount = 0;
    errorMessage = "";
    selectInput("Select a G-code file to process:", "fileSelected", null);
  }
  
  if (key == 'x' || key == 'X') {
    streaming = false;
    waitingResponse = false;
  }
  
  // Clear errors
  if (key == 'c' || key == 'C') {
    errorMessage = "";
    errorCount = 0;
  }
}

void fileSelected(File selection) {
  if (selection == null) {
    println("Window was closed or the user hit cancel.");
  } else {
    println("User selected " + selection.getAbsolutePath());
    gcode = loadStrings(selection.getAbsolutePath());
    if (gcode == null) return;
    streaming = true;
    i = 0;
    stream();
  }
}

void stream(){
  if (!streaming) return;
  
  while (true) {
    if (i == gcode.length) {
      streaming = false;
      return;
    }
    
    if (gcode[i].trim().length() == 0) i++;
    else break;
  }
  
  println("Sending: " + gcode[i]);
  port.write(gcode[i] + '\n');
  waitingResponse = true;
  lastResponseTime = millis();
  i++;
}

void serialEvent(Serial p){
  String s = p.readStringUntil('\n');
  if (s != null) {
    s = s.trim();
    lastResponse = s;
    waitingResponse = false;
    
    println("Received: " + s);
    
    if (s.startsWith("ok")) {
      okCount++;
      stream();
    } else if (s.startsWith("error")) {
      errorCount++;
      errorMessage = s;
      stream(); // Continue anyway or pause?
    } else {
      // Other responses - might be status from Arduino
      stream();
    }
  }
}
